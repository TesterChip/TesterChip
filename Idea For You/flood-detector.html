03 — SETUP

Step-by-Step Instructions

01

Install Arduino IDE

Download and install the Arduino IDE for your platform from arduino.cc/en/software. All three platforms are supported.

🪟 WINDOWS
arduino-ide-x.x.x-windows.exe
🍎 MACOS
arduino-ide-x.x.x-mac.dmg
🐧 LINUX
arduino-ide-x.x.x-linux.AppImage
02

Install Arduino Libraries

Open Arduino IDE → Sketch → Include Library → Manage Libraries and install:

  • LiquidCrystal I2C — by Frank de Brabander (for LCD display)
  • SD — built-in Arduino library (for SD card logging)
03

Install Python 3

Install Python 3 on your laptop. Choose your platform:

🪟 WINDOWS
python.org → Download
🍎 MACOS
brew install python3
🐧 LINUX
sudo apt install python3
04

Install Python Libraries

Open your terminal / command prompt and run:

TERMINAL 
pip install pyserial pandas scikit-learn
05

Find Your Arduino COM Port

Plug in your Arduino Uno Q via USB, then:

🪟 WINDOWS
Device Manager → COM Ports (e.g. COM3)
🍎 MACOS
ls /dev/tty.usb* (e.g. /dev/tty.usbmodem14101)
🐧 LINUX
ls /dev/ttyUSB* (e.g. /dev/ttyUSB0)

⚠ Replace COM3 in the Python script with your actual port.

06

Upload Arduino Sketch

Copy the Arduino sketch from the Code section below. In Arduino IDE:

  • Select Tools → Board → Arduino Uno
  • Select Tools → Port → your COM/tty port
  • Click the Upload ▶ button
  • Watch the LCD — it should show "Level: XX.X cm"
07

Run the Python AI Script

Save the Python script as flood_predictor.py and run it:

🪟 WINDOWS
python flood_predictor.py
🍎 MACOS / 🐧 LINUX
python3 flood_predictor.py

Keep the terminal open. After 10+ readings the AI prediction will appear automatically.

① Arduino Uno Q Sketch

Upload this to your board via Arduino IDE.

C++ / ARDUINO 
#include <LiquidCrystal_I2C.h>
#include <SD.h>

// LCD at I2C address 0x27, 16 columns, 2 rows
LiquidCrystal_I2C lcd(0x27, 16, 2);

const int trigPin     = 9;   // HC-SR04 Trigger
const int echoPin     = 10;  // HC-SR04 Echo
const int chipSelect  = 4;   // SD Card CS pin (onboard)

long  duration;
float distance;
float prevDistance = 0;
float slope        = 0;

void setup() {
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  Serial.begin(9600);

  lcd.init();
  lcd.backlight();

  // Initialize SD card
  if (!SD.begin(chipSelect)) {
    Serial.println("SD init failed! Check wiring.");
    return;
  }
  Serial.println("SD init success. Logging started.");

  // Write CSV header if file is new
  if (!SD.exists("floodlog.csv")) {
    File dataFile = SD.open("floodlog.csv", FILE_WRITE);
    if (dataFile) {
      dataFile.println("timestamp_ms,distance_cm,slope");
      dataFile.close();
    }
  }
}

void loop() {
  // ── 1. Measure distance via HC-SR04 ──────────────
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  duration = pulseIn(echoPin, HIGH);
  distance = duration * 0.034 / 2; // cm (speed of sound / 2)

  // ── 2. Calculate slope (rate of change) ──────────
  slope        = distance - prevDistance;
  prevDistance = distance;

  // ── 3. Display on LCD ────────────────────────────
  lcd.setCursor(0, 0);
  lcd.print("Level: ");
  lcd.print(distance, 1);
  lcd.print("cm   ");

  lcd.setCursor(0, 1);
  if      (slope > 0.5)  lcd.print("Rising  ^^  ");
  else if (slope < -0.5) lcd.print("Falling vv  ");
  else                   lcd.print("Stable  ==  ");

  // ── 4. Log to onboard SD card ────────────────────
  File dataFile = SD.open("floodlog.csv", FILE_WRITE);
  if (dataFile) {
    dataFile.print(millis());   // timestamp in ms
    dataFile.print(",");
    dataFile.print(distance, 2);
    dataFile.print(",");
    dataFile.println(slope, 2);
    dataFile.close();
  }

  // ── 5. Stream to laptop via USB serial ───────────
  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.print(" cm, Slope: ");
  Serial.println(slope);

  delay(10000); // poll every 10 seconds (adjust as needed)
}

② Python AI Prediction Script

Run on your laptop. Replace COM3 with your port.

PYTHON 3 
"""
flood_predictor.py — Arduino Uno Q Flood Detector
Reads serial data, loads SD-logged CSV, and predicts future water levels
using linear regression (scikit-learn).
"""

import serial
import pandas as pd
from   sklearn.linear_model import LinearRegression
import time
import sys
import os

# ── Configuration ─────────────────────────────────────────────────────────
# Windows :  'COM3'           (e.g. COM4, COM5 …)
# macOS   :  '/dev/tty.usbmodem14101'
# Linux   :  '/dev/ttyUSB0'  or  '/dev/ttyACM0'
SERIAL_PORT  = 'COM3'
BAUD_RATE    = 9600
LOG_FILE     = 'floodlog.csv'
FLOOD_THRESH = 100   # cm — adjust for your sensor height
MIN_ROWS     = 10   # minimum data points before prediction kicks in
PREDICT_STEPS= 5    # how many steps ahead to predict

# ── Serial connection ──────────────────────────────────────────────────────
def connect_serial(port, baud):
    try:
        ser = serial.Serial(port, baud, timeout=15)
        print(f"[OK] Connected to {port} at {baud} baud")
        return ser
    except serial.SerialException as e:
        print(f"[ERR] Could not open port {port}: {e}")
        print("      Check your port name and try again.")
        sys.exit(1)

# ── AI prediction ──────────────────────────────────────────────────────────
def predict_water_level(log_file, steps_ahead):
    try:
        df = pd.read_csv(log_file)
    except FileNotFoundError:
        print("[INFO] Log file not found yet — waiting for Arduino data...")
        return None

    if len(df) < MIN_ROWS:
        print(f"[INFO] Only {len(df)} rows logged — need {MIN_ROWS} to predict.")
        return None

    # Use row index as time feature (simple but effective for trend detection)
    X = df.index.values.reshape(-1, 1)
    y = df['distance_cm'].values

    model = LinearRegression()
    model.fit(X, y)

    future_idx = [[len(df) + steps_ahead]]
    predicted  = model.predict(future_idx)[0]
    return predicted

# ── Main loop ──────────────────────────────────────────────────────────────
def main():
    print("=" * 52)
    print("  Arduino Uno Q — Flood Detector AI Predictor")
    print("=" * 52)

    ser = connect_serial(SERIAL_PORT, BAUD_RATE)

    while True:
        try:
            raw  = ser.readline().decode('utf-8', errors='ignore').strip()
            if raw:
                print(f"\n[Arduino] {raw}")

            # Run prediction every 5 s regardless of serial pace
            predicted = predict_water_level(LOG_FILE, PREDICT_STEPS)

            if predicted is not None:
                print(f"[AI]      Predicted level in ~{PREDICT_STEPS * 10}s: {predicted:.1f} cm")
                if predicted > FLOOD_THRESH:
                    print(f"[ALERT]   ** FLOOD RISK — predicted {predicted:.1f} cm > {FLOOD_THRESH} cm **")
                elif predicted > FLOOD_THRESH * 0.8:
                    print(f"[WARN]    Water rising — approaching threshold ({FLOOD_THRESH} cm)")

        except KeyboardInterrupt:
            print("\n[INFO] Stopped by user.")
            ser.close()
            break
        except Exception as e:
            print(f"[ERR] {e}")

        time.sleep(5)

if __name__ == '__main__':
    main()
