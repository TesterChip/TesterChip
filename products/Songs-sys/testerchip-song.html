IMissYou_Song.ino Arduino UNO 
// ════════════════════════════════════════════
//   "I Miss You Come Here" — TesterChip
//   Arduino Uno · LCD I2C · Passive Buzzer
// ════════════════════════════════════════════

#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);
#define BUZZER_PIN  8
#define BPM         70
#define BEAT        (60000/BPM)
#define W  (BEAT*4)   #define H  (BEAT*2)
#define DQ (BEAT*3/2) #define Q  (BEAT)
#define E  (BEAT/2)

// ── Frequencies ──────────────────────────
#define REST 0    #define E4 330   #define G4 392
#define A4  440   #define B4 494   #define C5 523
#define D5  587   #define E5 659

struct Note { int freq, dur; };

Note melody[] = {
  // "I miss you, come here"
  {A4,E},{A4,E},{C5,Q},{B4,E},
  {A4,E},{G4,Q},{A4,H},
  // "And oh"
  {REST,E},{E4,Q},{G4,DQ},{REST,Q},
  // "It's hard to see you"
  {A4,E},{A4,E},{C5,Q},{B4,E},{A4,H},
  // "But I wish you"
  {G4,Q},{A4,Q},{B4,Q},{C5,Q},
  // "Were right here"
  {B4,Q},{A4,Q},{G4,H},{REST,Q},
  // "Oh! It's hard to leave you"
  {E4,E},{G4,E},{A4,Q},{B4,E},
  {C5,E},{B4,Q},{A4,H},
  // "When I get you everywhere"
  {G4,E},{A4,E},{B4,Q},{C5,E},
  {D5,E},{C5,Q},{B4,H},{REST,Q},
  // "All this time I'm thinking"
  {A4,E},{B4,E},{C5,Q},{B4,E},
  {A4,E},{G4,Q},
  // "I'm strong enough"
  {E4,Q},{G4,Q},{A4,Q},{B4,Q},
  // "To sink in"
  {C5,Q},{B4,Q},{A4,H},{REST,Q},
  // "Oh no!"
  {E5,E},{E5,E},{D5,H},{REST,E},
  // "I don't need you"
  {C5,Q},{B4,Q},{A4,Q},{G4,E},{A4,Q},
  // "But I miss you come here :("
  {A4,E},{A4,E},{C5,Q},{B4,E},
  {A4,E},{G4,Q},{E4,DQ},{REST,E},
  // Outro
  {A4,H},{G4,H},{E4,W},{REST,W},
};

struct Lyric { int idx; const char* r0; const char* r1; };
Lyric lyrics[] = {
  { 0, "I miss you",     "come here..."  },
  { 7, "And oh~",         ""              },
  {11, "It's hard to",   "see you..."    },
  {16, "But I wish you", ""              },
  {20, "Were right here",""              },
  {24, "Oh! It's hard",  "to leave you"  },
  {31, "When I get you", "everywhere"    },
  {39, "All this time",  "I'm thinking.."},
  {45, "I'm strong",      "enough"        },
  {49, "To sink in~",    ""              },
  {53, "Oh no!",          ""              },
  {57, "I don't need u", ""              },
  {62, "But I miss you", "come here :("  },
  {70, "~ ~ ~ ~ ~ ~",    ""              },
};

void lcdPrint(const char* top, const char* bot){
  lcd.clear();
  lcd.setCursor(0,0); lcd.print(top);
  lcd.setCursor(0,1); lcd.print(bot);
}

void playSong() {
  int N = sizeof(melody)/sizeof(melody[0]);
  int L = sizeof(lyrics)/sizeof(lyrics[0]);
  int li = 0;
  for(int i=0; i<N; i++){
    if(li<L && lyrics[li].idx==i)
      lcdPrint(lyrics[li].r0, lyrics[li++].r1);
    if(melody[i].freq != REST)
      tone(BUZZER_PIN, melody[i].freq,
           melody[i].dur*0.88);
    delay(melody[i].dur);
    noTone(BUZZER_PIN);
  }
}

void setup(){
  lcd.init(); lcd.backlight();
  pinMode(BUZZER_PIN, OUTPUT);
  playSong();
}
void loop(){} // press RESET to replay
